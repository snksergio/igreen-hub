// Medidor em meia-lua, no estilo de score: um segmento por fatura da sequência (meta de exemplo)
function medidor(feito, total){
  const cx = 110, cy = 104, r = 86, seg = 180 / total, folga = 3;
  const ponto = g => [cx + r * Math.cos(g * Math.PI / 180), cy - r * Math.sin(g * Math.PI / 180)];
  const cores = ["#F2D06B", "#EBD877", "#DDDE88", "#CFE39C", "#C3E9B1", "#B8EEC5"];
  let arcos = "";
  for (let i = 0; i < total; i++){
    const [x0, y0] = ponto(180 - i * seg - folga), [x1, y1] = ponto(180 - (i + 1) * seg + folga);
    arcos += `<path d="M${x0.toFixed(1)} ${y0.toFixed(1)} A${r} ${r} 0 0 1 ${x1.toFixed(1)} ${y1.toFixed(1)}" stroke="${i < feito ? cores[i] : "rgba(255,255,255,.2)"}" stroke-width="14" stroke-linecap="round" fill="none"/>`;
  }
  const [mx, my] = ponto(180 - feito * seg);
  return `<svg width="220" height="122" viewBox="0 0 220 122" aria-hidden="true">${arcos}
    ${feito ? `<circle cx="${mx.toFixed(1)}" cy="${my.toFixed(1)}" r="9" fill="#fff" stroke="#1C7443" stroke-width="4"/>` : ""}</svg>`;
}

function abaEconomia(d, e, sel){
  const ene = [14.10, 17.40, 20.20, 18.90, 21.50, st.pago ? 22.80 : 0];
  const max = Math.max(...e), media = d.eco / 6, mi = e.indexOf(max), ALT = 76;
  return `
    <div class="hb-graf" role="group" aria-label="Economia mês a mês, por produto">
      <span class="hb-media" style="bottom:${Math.round(media / max * ALT) + 22}px"><span>Média ${brl(media)}</span></span>
      ${e.map((v, i) => {
        const agora = i === 5 && !st.pago;
        return `<button type="button" data-mes-b="${i}" aria-pressed="${sel === i}" aria-label="${NOMES[i]}: ${brl(v)}, sendo energia ${brl(ene[i])} e Telecom ${brl(v - ene[i])}">
          <span class="pilha2 ${agora ? "agora" : ""}"><i class="ene" style="height:${Math.round(ene[i] / max * ALT)}px"></i><i class="tel" style="height:${Math.round((v - ene[i]) / max * ALT)}px"></i></span><span class="m">${MESES[i]}</span></button>`;
      }).join("")}

    <div class="hb-leg num hb-leg-graf"><span><i class="ene"></i>Energia</span><span><i class="tel"></i>Telecom</span></div>
    <div class="hb-kpis num">
