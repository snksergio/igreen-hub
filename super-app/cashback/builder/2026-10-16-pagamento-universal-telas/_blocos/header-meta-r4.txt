// Medidor em meia-lua, no estilo de score: um segmento por fatura da sequência (meta de exemplo)
const CORES_ESC = ["#F2D06B", "#EBD877", "#DDDE88", "#CFE39C", "#C3E9B1", "#B8EEC5"];
const CORES_CLARO = ["#F2D06B", "#E3D772", "#C9D97C", "#A6D388", "#86C97A", "#6BBF59"];
function medidor(feito, total, o = {}){
  const w = o.w || 220, h = o.h || 122, r = o.r || 86, sw = o.sw || 14, cx = w / 2, cy = h - (o.base ?? 18), seg = 180 / total, folga = o.folga ?? 3;
  const ponto = g => [cx + r * Math.cos(g * Math.PI / 180), cy - r * Math.sin(g * Math.PI / 180)];
  const cores = o.cores || CORES_ESC;
  let arcos = "";
  for (let i = 0; i < total; i++){
    const [x0, y0] = ponto(180 - i * seg - folga), [x1, y1] = ponto(180 - (i + 1) * seg + folga);
    arcos += `<path d="M${x0.toFixed(1)} ${y0.toFixed(1)} A${r} ${r} 0 0 1 ${x1.toFixed(1)} ${y1.toFixed(1)}" stroke="${i < feito ? cores[i] : (o.vazio || "rgba(255,255,255,.2)")}" stroke-width="${sw}" stroke-linecap="round" fill="none"/>`;
  }
  const [mx, my] = ponto(180 - feito * seg);
  return `<svg width="${w}" height="${h}" viewBox="0 0 ${w} ${h}" aria-hidden="true">${arcos}
    ${feito ? `<circle cx="${mx.toFixed(1)}" cy="${my.toFixed(1)}" r="${(sw * .62).toFixed(1)}" fill="#fff" stroke="#1C7443" stroke-width="3"/>` : ""}</svg>`;
}
// Cartão da meta: medidor à esquerda, descrição à direita
function cartaoMeta(d, claro){
  const bateu = d.emDia >= 6;
  const g = medidor(d.emDia, 6, {w:132, h:78, r:54, sw:10, base:10, cores: claro ? CORES_CLARO : CORES_ESC, vazio: claro ? "#E7E3D8" : "rgba(255,255,255,.2)"});
  return `<div class="meta-card ${claro ? "claro" : ""} ${bateu ? "pulsa" : ""}">
    <div class="meta-g" role="img" aria-label="${d.emDia} de 6 faturas seguidas pagas em dia. Situação: saudável. Meta de exemplo: 6">${g}<div class="meta-n num"><b>${d.emDia}</b><small>de 6</small></div></div>
    <div class="meta-t">
      <span class="chip saude">${svg(I.check,12,3.5)} Saudável</span>
      <b>${bateu ? "Meta de exemplo batida" : `${d.emDia} faturas seguidas em dia`}</b>
      <span>${bateu ? "6 faturas seguidas pagas em dia" : "Meta de exemplo: 6 · outubro vence em 23/10"}</span>
      <div class="bolas" aria-hidden="true">${MESES.map((_, i) => `<i class="${i < d.emDia ? "ok" : (i === 5 ? "agora" : "")}"></i>`).join("")}</div>
    </div>
  </div>`;
}
